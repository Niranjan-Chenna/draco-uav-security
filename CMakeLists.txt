cmake_minimum_required(VERSION 3.16)
project(draco LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
set(MAVLINK_INCLUDE_DIR "$ENV{HOME}/PX4-Autopilot/build/px4_sitl_default/mavlink" CACHE PATH "generated MAVLink headers")
if(NOT EXISTS "${MAVLINK_INCLUDE_DIR}/development/mavlink.h")
  message(FATAL_ERROR "Set MAVLINK_INCLUDE_DIR to generated PX4 MAVLink headers")
endif()
find_package(OpenSSL REQUIRED)
add_library(draco_core canonical_mission.cpp mavlink_parser.cpp mission_authorization.cpp
  mission_change_budget.cpp mission_decision_record.cpp mission_delta.cpp
  mission_intent_contract.cpp mission_reconstructor.cpp mission_revision.cpp
  mission_revision_causality.cpp mission_revision_tracker.cpp phase4_research_freeze.cpp
  semantic_classifier.cpp state_cache.cpp runtime_policy.cpp principal_context.cpp evaluation_mode.cpp
  mission_pipeline.cpp structured_events.cpp gateway_ack_reconciliation.cpp)
target_include_directories(draco_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(draco_core SYSTEM PUBLIC ${MAVLINK_INCLUDE_DIR})
target_link_libraries(draco_core PUBLIC OpenSSL::Crypto)
add_executable(draco main.cpp udp_gateway.cpp)
target_link_libraries(draco PRIVATE draco_core)
add_library(evaluation_scenarios evaluation/scenarios.cpp)
target_link_libraries(evaluation_scenarios PUBLIC draco_core)
add_executable(core_runner evaluation/core_runner.cpp)
target_link_libraries(core_runner PRIVATE evaluation_scenarios)
add_executable(mission_client evaluation/mission_client.cpp)
target_link_libraries(mission_client PRIVATE evaluation_scenarios)
add_executable(delta_accuracy evaluation/delta_accuracy.cpp)
target_link_libraries(delta_accuracy PRIVATE evaluation_scenarios)
enable_testing()
file(GLOB test_sources CONFIGURE_DEPENDS tests/*.cpp)
foreach(source ${test_sources})
  get_filename_component(name ${source} NAME_WE)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE draco_core)
  target_compile_options(${name} PRIVATE -UNDEBUG)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endforeach()
target_link_libraries(evaluation_scenarios_test PRIVATE evaluation_scenarios)
add_dependencies(gateway_transport_test draco)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME observer_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/observer_test.py)
